        function [Phase0,CameraRes] = readZygoDat(app)
            fullfilename0=app.fullfilenameG;%读取文件名称
            fid = fopen(fullfilename0,'r','b');
            Magic_Number = fread(fid,1,'int32');
            header_format = fread(fid,1,'int16');
            fseek(fid,60,'bof');
            NumberOfBytesIntensity = fread(fid,1,'int32');
            fseek(fid,64,'bof');
            PhaseOriginX = fread(fid,1,'int16');
            PhaseOriginY = fread(fid,1,'int16');
            PhaseWidth = fread(fid,1,'int16');
            PhaseHeight = fread(fid,1,'int16');
            fseek(fid,164,"bof");
            intf_scale_factor = fread(fid,1,'float32');
            wavelength_in = fread(fid,1,'float32');
            fseek(fid,176,"bof");
            obliquity_factor = fread(fid,1,'float32');
            fseek(fid,184,'bof');
            CameraRes = fread(fid,1,'float32')*1000;
            fseek(fid,218,'bof');
            PhaseRes = fread(fid,1,'int16');
            if PhaseRes == 0
                PhaseRes = 4096;
            elseif PhaseRes == 1
                PhaseRes = 32768;
            end
            fseek(fid,234,'bof');
            CameraWidth = fread(fid,1,'int16');
            CameraHeight = fread(fid,1,'int16');
            coef = intf_scale_factor*obliquity_factor/PhaseRes;
            if header_format~=3
                fseek(fid,NumberOfBytesIntensity+834,'bof');
            else
                fseek(fid,NumberOfBytesIntensity+4096,'bof');
            end
            data = fread(fid,[PhaseWidth,PhaseHeight],'int32');
            data = data';
            Phase0 = data;
            mask0 = true(size(Phase0));
            mask0(Phase0>=2147483640) = false;
            Phase0(~mask0) = nan;
            Phase0 = Phase0*coef;
            Phase = nan(CameraHeight,CameraWidth);
            Phase(PhaseOriginY+1:PhaseOriginY+PhaseHeight, PhaseOriginX+1:PhaseOriginX+PhaseWidth) = Phase0;
            fclose(fid);
        end
        
        function z = Asphere(app)
            x=app.xG;%用于计算面型的通用数据
            y=app.yG;%用于计算面型的通用数据
            R=app.RG;%用于计算面型的通用数据
            K=app.KG;%用于计算面型的通用数据
            An=app.AnG;%用于计算面型的通用数据
            rp=sqrt(x.^2+y.^2);%样本点母镜坐标
            if R~=0
                c=1/R;%样本点母镜坐标
                z=-real(c*rp.^2./(1+sqrt(1-(K+1)*c^2*rp.^2)));%样本点母镜坐标
            elseif R==0
                z=zeros(size(x));%曲率半径为0，则是平面
            end
            for iA=1:length(An)
                if An(iA)~=0||~isnan(An(iA))
                    z=z-An(iA).*rp.^(2*(iA+1));%高次项叠加
                end
            end
        end
        
        function z = BiCurve(app)
            x=app.xG;%用于计算面型的通用数据
            y=app.yG;%用于计算面型的通用数据
            Rx=app.RxG;%用于计算面型的通用数据
            Ry=app.RyG;%用于计算面型的通用数据
            Kx=app.KxG;%用于计算面型的通用数据
            Ky=app.KyG;%用于计算面型的通用数据
            Ax=app.AxG;%用于计算面型的通用数据
            Ay=app.AyG;%用于计算面型的通用数据
            if Rx~=0 && Ry~=0
                cx=1/Rx;%样本点母镜坐标
                cy=1/Ry;%样本点母镜坐标
                z=-real((cx*x.^2+cy*y.^2)./(1+sqrt(1-(Kx+1)*cx^2*x.^2-(Ky+1)*cy^2*y.^2)));%样本点母镜坐标z
            elseif Rx~=0 && Ry==0
                cx=1/Rx;%样本点母镜坐标
                z=-real(cx*x.^2./(1+sqrt(1-(Kx+1)*cx^2*x.^2)));%样本点母镜坐标zx
            elseif Rx==0 && Ry~=0
                cy=1/Ry;%样本点母镜坐标
                z=-real(cy*y.^2./(1+sqrt(1-(Ky+1)*cy^2*y.^2)));%样本点母镜坐标zy
            elseif Rx==0 && Ry==0
                z=zeros(size(x));%曲率半径为0，平面
            end
            for iAx=1:length(Ax)
                if Ax(iAx)~=0||~isnan(Ax(iAx))
                    z=z-Ax(iAx).*x.^(2*(iAx+1));%高次项叠加
                end
            end
            for iAy=1:length(Ay)
                if Ay(iAy)~=0||~isnan(Ay(iAy))
                    z=z-Ay(iAy).*y.^(2*(iAy+1));%高次项叠加
                end
            end
        end

                cy=1/Ry;%样本点母镜坐标
                z=-real((cx*x.^2+cy*y.^2)./(1+sqrt(1-(Kx+1)*cx^2*x.^2-(Ky+1)*cy^2*y.^2)));%样本点母镜坐标z
            elseif Rx~=0 && Ry==0
                cx=1/Rx;%样本点母镜坐标
                z=-real(cx*x.^2./(1+sqrt(1-(Kx+1)*cx^2*x.^2)));%样本点母镜坐标zx
            elseif Rx==0 && Ry~=0
                cy=1/Ry;%样本点母镜坐标
                z=-real(cy*y.^2./(1+sqrt(1-(Ky+1)*cy^2*y.^2)));%样本点母镜坐标zy
            elseif Rx==0 && Ry==0
                z=zeros(size(x));%曲率半径为0，平面
            end
            z=z-x.^1.*y.^0*Axy(1)- x.^0.*y.^1*Axy(2)...
               -x.^2.*y.^0*Axy(3)- x.^1.*y.^1*Axy(4)- x.^0.*y.^2*Axy(5)...
               -x.^3.*y.^0*Axy(6)- x.^2.*y.^1*Axy(7)- x.^1.*y.^2*Axy(8)- x.^0.*y.^3*Axy(9)...
               -x.^4.*y.^0*Axy(10)-x.^3.*y.^1*Axy(11)-x.^2.*y.^2*Axy(12)-x.^1.*y.^3*Axy(13)-x.^0.*y.^4*Axy(14)...
               -x.^5.*y.^0*Axy(15)-x.^4.*y.^1*Axy(16)-x.^3.*y.^2*Axy(17)-x.^2.*y.^3*Axy(18)-x.^1.*y.^4*Axy(19)-x.^0.*y.^5*Axy(20)...
               -x.^6.*y.^0*Axy(21)-x.^5.*y.^1*Axy(22)-x.^4.*y.^2*Axy(23)-x.^3.*y.^3*Axy(24)-x.^2.*y.^4*Axy(25)-x.^1.*y.^5*Axy(26)-x.^0.*y.^6*Axy(27)...
               -x.^7.*y.^0*Axy(28)-x.^6.*y.^1*Axy(29)-x.^5.*y.^2*Axy(30)-x.^4.*y.^3*Axy(31)-x.^3.*y.^4*Axy(32)-x.^2.*y.^5*Axy(33)-x.^1.*y.^6*Axy(34)-x.^0.*y.^7*Axy(35)...
               -x.^8.*y.^0*Axy(36)-x.^7.*y.^1*Axy(37)-x.^6.*y.^2*Axy(38)-x.^5.*y.^3*Axy(39)-x.^4.*y.^4*Axy(40)-x.^3.*y.^5*Axy(41)-x.^2.*y.^6*Axy(42)-x.^1.*y.^7*Axy(43)-x.^0.*y.^8*Axy(44)...
               -x.^9.*y.^0*Axy(45)-x.^8.*y.^1*Axy(46)-x.^7.*y.^2*Axy(47)-x.^6.*y.^3*Axy(48)-x.^5.*y.^4*Axy(49)-x.^4.*y.^5*Axy(50)-x.^3.*y.^6*Axy(51)-x.^2.*y.^7*Axy(52)-x.^1.*y.^8*Axy(53)-x.^0.*y.^9*Axy(54);
        end
    end

            z=reshape(z2d,1,Nx0*Ny0);%向量化用于坐标变换
            nx=reshape(nx2d,1,Nx0*Ny0);%向量化用于坐标变换
            ny=reshape(ny2d,1,Nx0*Ny0);%向量化用于坐标变换
            nz=reshape(nz2d,1,Nx0*Ny0);%向量化用于坐标变换
            if betB==0 %如果没有倾斜角，则需不要旋转计算
                xcln0=x0;%x坐标
                ycln0=y0;%y坐标
                zcln0=z;%z坐标
                nxcln0=nx;%nx
                nycln0=ny;%ny
                nzcln0=nz;%nz
            else
                Tbet=[cos(betB),0,sin(betB);0,1,0;-sin(betB),0,cos(betB)];%倾斜角拉平
                Tgam=[cos(gamB),-sin(gamB),0;sin(gamB),cos(gamB),0;0,0,1];%近远轴方向旋转
                TgamN=[cos(-gamB),-sin(-gamB),0;sin(-gamB),cos(-gamB),0;0,0,1];%近远轴方向旋转
                pxyz=Tgam*Tbet*TgamN*[x0;y0;z];%变换
                nxyz=Tgam*Tbet*TgamN*[nx;ny;nz];%变换
                xcln0=pxyz(1,:);%x坐标
                ycln0=pxyz(2,:);%y坐标
                zcln0=pxyz(3,:);%z坐标
                nxcln0=nxyz(1,:);%nx
                nycln0=nxyz(2,:);%ny
                nzcln0=nxyz(3,:);%nz
            end
            %工件口径点云
            if strcmp(app.ApertureTabGroup.SelectedTab.Title,'圆口径')
                D_suf=app.DiamEditField.Value;%点云口径
                xmld=xcln0(xcln0.^2+ycln0.^2<(D_suf/2)^2); %模型坐标
                ymld=ycln0(xcln0.^2+ycln0.^2<(D_suf/2)^2); %模型坐标
                zmld=zcln0(xcln0.^2+ycln0.^2<(D_suf/2)^2); %模型坐标
                nxmld=nxcln0(xcln0.^2+ycln0.^2<(D_suf/2)^2); %模型坐标
                nymld=nycln0(xcln0.^2+ycln0.^2<(D_suf/2)^2); %模型坐标
                nzmld=nzcln0(xcln0.^2+ycln0.^2<(D_suf/2)^2); %模型坐标
            elseif strcmp(app.ApertureTabGroup.SelectedTab.Title,'矩形')
                Lx=app.LXEditField.Value;%Lx
                Ly=app.LYEditField.Value;%Ly
                Rcor=app.RCornerEditField.Value;
                index1=false(size(xcln0));%mask标识
                index1(abs(xcln0)>(Lx/2-Rcor))=true;%位于角部
                index2=false(size(xcln0));%mask标识
                index2(abs(ycln0)>(Ly/2-Rcor))=true;%位于角部
                index3=false(size(xcln0));%mask标识
                index3((abs(xcln0)-(Lx/2-Rcor)).^2+(abs(ycln0)-(Ly/2-Rcor)).^2>Rcor^2)=true;%位于角部且大于导圆半径
                index4=false(size(xcln0));%mask标识
                index4(abs(xcln0)>(Lx/2))=true;%位于角外部
                index5=false(size(xcln0));%mask标识
                index5(abs(ycln0)>(Ly/2))=true;%位于角外部
                index=~(index1&index2&index3|index4|index5);%外部区域取反为内部区域
                xmld=xcln0(index);%口径外剔除
                ymld=ycln0(index);%口径外剔除
                zmld=zcln0(index);%口径外剔除
                nxmld=nxcln0(index);%口径外剔除
                nymld=nycln0(index);%口径外剔除
                nzmld=nzcln0(index);%口径外剔除
            elseif strcmp(app.ApertureTabGroup.SelectedTab.Title,'正多边形')
                N_sides=app.EdgeNumEditField.Value;%顶点数
                D=app.DiamPolyEditField.Value;
                tht0=(2*pi)/360*app.FstVerAngNumEditField.Value;%第一角的角位
                N_vertx=N_sides;%顶点数
                index=true(size(xcln0));%mask
                tht_vertx=linspace(tht0,2*pi+tht0,N_vertx+1);%顶点角度
                x_vertex=(D/2)*cos(tht_vertx);%顶点坐标
                y_vertex=(D/2)*sin(tht_vertx);%顶点坐标
                for inod=1:length(xcln0)
                    xp=xcln0(inod);%逐点扫描
                    yp=ycln0(inod);%逐点扫描
                    for i_sides=1:N_sides
                        n1=[x_vertex(i_sides)-xp;y_vertex(i_sides)-yp;0];%点到边起点矢量
                        n2=[x_vertex(i_sides+1)-x_vertex(i_sides);y_vertex(i_sides+1)-y_vertex(i_sides);0];%边矢量
                        n3=cross(n1,n2);%叉乘矢量
                        if n3(3)<0
                            index(1,inod)=false;%叉乘矢量为负，则在边界之内
                            break;
                        end
                    end
                end
                xmld=xcln0(index);%口径外剔除
                ymld=ycln0(index);%口径外剔除
                zmld=zcln0(index);%口径外剔除
                nxmld=nxcln0(index);%口径外剔除
                nymld=nycln0(index);%口径外剔除
                nzmld=nzcln0(index);%口径外剔除
            end
            xv1d=linspace(-D_suf/2,D_suf/2,101);%显示坐标
            yv1d=linspace(-D_suf/2,D_suf/2,101);%显示坐标
            [xv2d,yv2d]=meshgrid(xv1d,yv1d);%显示坐标
            if isempty(xmld)||isempty(ymld)||isempty(zmld)
                plot(app.UIAxes2D,xmld,ymld);%无点绘图
                plot3(app.UIAxes3D,xmld,ymld,zmld);%无点绘图
            else
                zv2d=griddata(xmld,ymld,zmld,xv2d,yv2d);%插值绘图
                if strcmp(app.ApertureTabGroup.SelectedTab.Title,'圆口径')
                    zv2d(xv2d.^2+yv2d.^2<(D_inner/2)^2)=NaN;%非数
                end
                surf(app.UIAxes2D,xv2d,yv2d,zv2d);%插值绘图
                shading(app.UIAxes2D,"interp");%插值绘图
                axis(app.UIAxes2D,'equal');%插值绘图
                colorbar(app.UIAxes2D);%插值绘图
                view(app.UIAxes2D,[0 0 1]);%插值绘图
                surf(app.UIAxes3D,xv2d,yv2d,zv2d,zv2d);%插值绘图
                shading(app.UIAxes3D,"interp");%插值绘图
                axis(app.UIAxes3D,'equal');%插值绘图
                app.ModelLamp.Color=[0.00,1.00,0.00];
            end
            %拓展区域点云--------------------------------------------------
            app.xmldG=xmld;
            app.ymldG=ymld;
            app.zmldG=zmld;
            app.nxmldG=nxmld;
            app.nymldG=nymld;
            app.nzmldG=nzmld;
            app.xmld0G=xcln0;
            app.ymld0G=ycln0;
            app.zmld0G=zcln0;
            app.nxmld0G=nxcln0;
            app.nymld0G=nycln0;
            app.nzmld0G=nzcln0;
            An=app.AnG;%通用高阶系数
            Ax=app.AxG;%X高阶系数
            Ay=app.AyG;%Y高阶系数
            Axy=app.AxyG;%交叉项系数
            if ~isempty(An)
                for iAn=1:length(An)
                    app.AnUITable.Data(iAn,2)=cellstr(num2str(An(iAn,1),10));%转换为数字
                end
            end
            if ~isempty(Ax)
                for iAx=1:length(Ax)
                    app.AxUITable.Data(iAx,2)=cellstr(num2str(Ax(iAx,1),10));%转换为数字
                end
            end
            if ~isempty(Ay)
                for iAy=1:length(Ay)
                    app.AyUITable.Data(iAy,2)=cellstr(num2str(Ay(iAy,1),10));%转换为数字
                end
            end
            if ~isempty(Axy)
                for iAxy=1:length(Axy)
                    app.FFUITable.Data(iAxy,2)=cellstr(num2str(Axy(iAxy,1),10));%转换为数字
                end
            end
            app.SagEditField.Value='';%矢高验算为空
            app.AngleEditField.Value='';%角度验算为空
        end

            axis(app.PostDataSurfUIAxes,'equal');%画图
            view(app.PostDataSurfUIAxes,[0 0 1]);%画图

            app.y2dSufPostG=app.y2dAfter2G;%处理后数据
        end

                W2dext(~index)=NaN;%考虑圆角的延拓，圆角外NaN
            elseif strcmp(app.ApertureTabGroup.SelectedTab.Title,'正多边形')              
                [Ny0,Nx0]=size(W2d0);%数据长宽
                dx=app.DXCalEditField.Value;%面形数据间距与设置间距一致
                dy=app.DYCalEditField.Value;%面形数据间距与设置间距一致
                D=app.DiamPolyEditField.Value;%直径
                Dext=D+2*Extd;%延拓后直径
                NxHalfExt=ceil(Dext/2/dx);%半节点数
                NyHalfExt=ceil(Dext/2/dy);%半节点数
                x1dext=dx*(-NxHalfExt:1:NxHalfExt);%坐标
                y1dext=dy*(-NyHalfExt:1:NyHalfExt);%坐标
                [x2dext,y2dext]=meshgrid(x1dext,y1dext);%坐标
                Wcln0=reshape(W2d0,Ny0*Nx0,1);%变换
                xcln0=reshape(x2d,Ny0*Nx0,1);%变换
                ycln0=reshape(y2d,Ny0*Nx0,1);%变换
                Wcln=Wcln0(~isnan(Wcln0));%剔除非数
                xcln=xcln0(~isnan(Wcln0));%剔除非数
                ycln=ycln0(~isnan(Wcln0));%剔除非数
                W2dext=griddata(xcln,ycln,Wcln,x2dext,y2dext,'nearest');%插值
                [Ny,Nx]=size(W2dext);%面形数量
                index=true(size(W2dext));%mask
                N_sides=app.EdgeNumEditField.Value;%顶点数
                N_vertx=N_sides;%顶点数
                tht0=(2*pi)/360*app.FstVerAngNumEditField.Value;%第一角的角位
                tht_vertx=linspace(tht0,2*pi+tht0,N_vertx+1);%顶点角度
                x_vertex=(Dext/2)*cos(tht_vertx);%顶点坐标
                y_vertex=(Dext/2)*sin(tht_vertx);%顶点坐标
                for jx=1:Nx
                    for iy=1:Ny
                        xp=x2dext(iy,jx);
                        yp=y2dext(iy,jx);
                        for i_sides=1:N_sides
                            n1=[x_vertex(i_sides)-xp;y_vertex(i_sides)-yp;0];%点到边起点矢量
                            n2=[x_vertex(i_sides+1)-x_vertex(i_sides);y_vertex(i_sides+1)-y_vertex(i_sides);0];%边矢量
                            n3=cross(n1,n2);%叉乘矢量
                            if n3(3)<0
                                index(iy,jx)=false;%叉乘矢量为负，则幅值false，则出边界，index为true是边界内，index为false是边界外
                                break;
                            end
                        end
                    end
                end
                W2dext(~index)=NaN;%index为true是边界内，index为false是边界外，边界外赋值NaN
            end
            [m,n]=size(W2dext);%矩阵大小
            if isequal(size(x2dext),size(W2dext)) && isequal(size(y2dext),size(W2dext)) && m>1 && n>1
                surf(app.PostDataSurfUIAxes,x2dext,y2dext,W2dext);%画图
                shading(app.PostDataSurfUIAxes,'flat');%画图
                axis(app.PostDataSurfUIAxes,'equal');%画图
                view(app.PostDataSurfUIAxes,[0 0 1]);%画图
                colorbar(app.PostDataSurfUIAxes);%画图
                app.W2dAfter2G=W2dext;%处理后数据
                app.x2dAfter2G=x2dext;%处理后数据
                app.y2dAfter2G=y2dext;%处理后数据
                app.SurfDatLamp.Color=[0.00,1.00,0.00];%显示灯
                app.W2dSufPostG=app.W2dAfter2G;%轨迹算面形
                app.x2dSufPostG=app.x2dAfter2G;%处理后数据
                app.y2dSufPostG=app.y2dAfter2G;%处理后数据
            end
        end

                index3((abs(x2dCal)-(LxTrj/2-RcorTrimTrj)).^2+(abs(y2dCal)-(LyTrj/2-RcorTrimTrj)).^2>RcorTrimTrj^2)=true;%位于角部且大于导圆半径
                index4=false(NyCal,NxCal);%mask标识
                index4(abs(x2dCal)>(LxTrj/2))=true;%位于角外部
                index5=false(NyCal,NxCal);%mask标识
                index5(abs(y2dCal)>(LyTrj/2))=true;%位于角外部
                MaskTrjCal=~(index1&index2&index3|index4|index5);%外部区域取反为内部区域
                %--------------------------------------------------------------------------
                index1=false(NyCal,NxCal);%mask标识
                index1(abs(x2dCal)>(LxEva/2-RcorTrimEva))=true;%位于角部
                index2=false(NyCal,NxCal);%mask标识
                index2(abs(y2dCal)>(LyEva/2-RcorTrimEva))=true;%位于角部
                index3=false(NyCal,NxCal);%mask标识
                index3((abs(x2dCal)-(LxEva/2-RcorTrimEva)).^2+(abs(y2dCal)-(LyEva/2-RcorTrimEva)).^2>RcorTrimEva^2)=true;%位于角部且大于导圆半径
                index4=false(NyCal,NxCal);%mask标识
                index4(abs(x2dCal)>(LxEva/2))=true;%位于角外部
                index5=false(NyCal,NxCal);%mask标识
                index5(abs(y2dCal)>(LyEva/2))=true;%位于角外部
                MaskEvaCal=~(index1&index2&index3|index4|index5);%外部区域取反为内部区域
            elseif strcmp(app.ApertureTabGroup.SelectedTab.Title,'正多边形') 
                TrimTrj=app.TrimTrjEditField.Value;%轨迹区裁边
                TrimEva=app.TrimEvaEditField.Value;%评价区裁边
                DSuf=app.DiamPolyEditField.Value;%直径
                DTrj=DSuf-2*TrimTrj;%轨迹区直径
                DEva=DSuf-2*TrimEva;%评价区直径
                MaskSufCal=true(NyCal,NxCal);%mask
                MaskTrjCal=true(NyCal,NxCal);%mask
                MaskEvaCal=true(NyCal,NxCal);%mask
                N_sides=app.EdgeNumEditField.Value;%顶点数
                N_vertx=N_sides;%顶点数
                tht0=(2*pi)/360*app.FstVerAngNumEditField.Value;%第一角的角位
                tht_vertx=linspace(tht0,2*pi+tht0,N_vertx+1);%顶点角度
                x_vertexSuf=(DSuf/2)*cos(tht_vertx);%顶点坐标
                y_vertexSuf=(DSuf/2)*sin(tht_vertx);%顶点坐标
                x_vertexTrj=(DTrj/2)*cos(tht_vertx);%顶点坐标
                y_vertexTrj=(DTrj/2)*sin(tht_vertx);%顶点坐标
                x_vertexEva=(DEva/2)*cos(tht_vertx);%顶点坐标
                y_vertexEva=(DEva/2)*sin(tht_vertx);%顶点坐标
                for jx=1:NxCal
                    for iy=1:NyCal
                        xp=x2dCal(iy,jx);
                        yp=y2dCal(iy,jx);
                        for i_sides=1:N_sides
                            n1=[x_vertexSuf(i_sides)-xp;y_vertexSuf(i_sides)-yp;0];%点到边起点矢量
                            n2=[x_vertexSuf(i_sides+1)-x_vertexSuf(i_sides);y_vertexSuf(i_sides+1)-y_vertexSuf(i_sides);0];%边矢量
                            n3=cross(n1,n2);%叉乘矢量
                            if n3(3)<0
                                MaskSufCal(iy,jx)=false;%叉乘矢量为负，则幅值false，则出边界，index为true是边界内，index为false是边界外
                                break;
                            end
                        end
                        for i_sides=1:N_sides
                            n1=[x_vertexTrj(i_sides)-xp;y_vertexTrj(i_sides)-yp;0];%点到边起点矢量
                            n2=[x_vertexTrj(i_sides+1)-x_vertexTrj(i_sides);y_vertexTrj(i_sides+1)-y_vertexTrj(i_sides);0];%边矢量
                            n3=cross(n1,n2);%叉乘矢量
                            if n3(3)<0
                                MaskTrjCal(iy,jx)=false;%叉乘矢量为负，则幅值false，则出边界，index为true是边界内，index为false是边界外
                                break;
                            end
                        end
                        for i_sides=1:N_sides
                            n1=[x_vertexEva(i_sides)-xp;y_vertexEva(i_sides)-yp;0];%点到边起点矢量
                            n2=[x_vertexEva(i_sides+1)-x_vertexEva(i_sides);y_vertexEva(i_sides+1)-y_vertexEva(i_sides);0];%边矢量
                            n3=cross(n1,n2);%叉乘矢量
                            if n3(3)<0
                                MaskEvaCal(iy,jx)=false;%叉乘矢量为负，则幅值false，则出边界，index为true是边界内，index为false是边界外
                                break;
                            end
                        end
                    end
                end
            end
            %--------------------------------------------------------------
            W2d0PstCal=nan(NyCal,NxCal);%面形
            W2d0PstCal(NyMidCal+(-NyHalfPst:NyHalfPst),NxMidCal+(-NxHalfPst:NxHalfPst))=W2d0Pst;%面形
            selectedButton = app.SolveModeButtonGroup.SelectedObject;
            Impulse=app.ImpulseG;%脉冲去除率
            UniformThick=app.UniformThickEditField.Value;%均抛层厚度
            MaxDwell0=app.MaxDwellTimeEditField.Value;%最大驻留时间
            MinDwell0=app.MinDwellTimeEditField.Value;%最小驻留时间
            MaxDwell=MaxDwell0;%
            MinDwell=MinDwell0;%
            if strcmp(selectedButton.Text,'脉冲迭代法')==1    
                NIter=app.IterTimeEditField.Value;%迭代次数
                WN2d0PstCal=W2d0PstCal-min(min(W2d0PstCal));%待加工量是高出最低点的高度
                WN2d0PstCal=WN2d0PstCal+UniformThick;%待加工量要加上均抛层厚度
                WN2d0PstCal(isnan(WN2d0PstCal))=0;%非数区待加工量为0
                DwellTimeCal0=WN2d0PstCal/Impulse;%驻留时间Cal
                dDwellTime=zeros(NyCal,NxCal);%驻留时间修正
                for iter=1:NIter
                    WP2dPstCal=zeros(NyCal,NxCal);%实际加工量仿真
                    DwellTimeCal0=DwellTimeCal0+dDwellTime;%驻留时间修正
                    DwellTimeCal0(DwellTimeCal0<MinDwell)=MinDwell;%驻留时间不能小于0
                    DwellTimeCal0(DwellTimeCal0>MaxDwell)=MaxDwell;%驻留时间不能小于0
                    DwellTimeCal0(~MaskTrjCal)=NaN;%驻留时间Trj
                    for iy=NyMidCal+(-NyHalfPst:NyHalfPst)
                        for jx=NxMidCal+(-NxHalfPst:NxHalfPst)
                            if DwellTimeCal0(iy,jx)>0 && ~isnan(DwellTimeCal0(iy,jx))
                                WP2dPstCal(iy+(-NySptHalfDwn:NySptHalfUp),jx+(-NxSptHalfLft:NxSptHalfRgt))=...
                                    WP2dPstCal(iy+(-NySptHalfDwn:NySptHalfUp),jx+(-NxSptHalfLft:NxSptHalfRgt))+...
                                    DwellTimeCal0(iy,jx)*W2dSptRmv(NyMidSpt+(-NySptHalfDwn:NySptHalfUp),NxMidSpt+(-NxSptHalfLft:NxSptHalfRgt));
                            end
                        end
                    end
                    WRsd2dPstCal=W2d0PstCal-WP2dPstCal;%残差
                    WRsd2dPstCal(isnan(WRsd2dPstCal))=0;%非数去没有残差
                    WRsd2dPstCal=WRsd2dPstCal-min(min(WRsd2dPstCal));%残差
                    dDwellTime=WRsd2dPstCal/Impulse;%驻留时间修正
                end
            elseif strcmp(selectedButton.Text,'最小二乘法')==1
                WN2d0PstCal=W2d0PstCal-min(min(W2d0PstCal));%待加工量是高出最低点的高度
                WN2d0PstCal=WN2d0PstCal+UniformThick;%待加工量要加上均抛层厚度
                WNCln0PstCal=WN2d0PstCal(:);%剔除非数
                NumNodW=length(WNCln0PstCal(~isnan(WNCln0PstCal)));%非零节点数
                if NumNodW>9500
                    msgbox('不适合最小二乘法，计算量大');
                else
                    NNodTrj=sum(sum(MaskTrjCal));%节点总数
                    Ser2dTrj=zeros(NyCal,NxCal);%mask
                    Ser2dTrj(MaskTrjCal)=(1:NNodTrj)';%全局序号
                    NNodEva=sum(sum(MaskEvaCal));%节点总数
                    Ser2dEva=zeros(NyCal,NxCal);%mask
                    Ser2dEva(MaskEvaCal)=(1:NNodEva)';%全局序号
                    NySpt=NySptHalfDwn+NySptHalfUp+1;%抛光斑节点数
                    NxSpt=NxSptHalfLft+NxSptHalfRgt+1;%抛光斑节点数
                    W2dSptRmv2=W2dSptRmv(NyMidSpt+(-NySptHalfDwn:NySptHalfUp),NxMidSpt+(-NxSptHalfLft:NxSptHalfRgt));%只提取有数据的部分
                    WClnSptRmv=W2dSptRmv2(:);%去除函数向量化
                    AConvEva=sparse(NNodEva,NNodTrj);%卷积矩阵
                    for iy=NyMidCal+(-NyHalfPst:NyHalfPst)
                        for jx=NxMidCal+(-NxHalfPst:NxHalfPst)
                            if MaskTrjCal(iy,jx)
                                Ser2dEvaIn=Ser2dEva(iy+(-NySptHalfDwn:NySptHalfUp),jx+(-NxSptHalfLft:NxSptHalfRgt));%Eva的序号落在抛光斑内的序号
                                SerClnEvaIn=Ser2dEvaIn(:);%Eva的序号落在抛光斑内的序号，向量化
                                Idx2dEvaIn=MaskEvaCal(iy+(-NySptHalfDwn:NySptHalfUp),jx+(-NxSptHalfLft:NxSptHalfRgt));%Eva的Mask落在抛光斑内
                                IdxClnEvaIn=Idx2dEvaIn(:);%Eva的Mask落在抛光斑内，列向量化
                                SerClnEvaInNN=SerClnEvaIn(IdxClnEvaIn);%Eva序号落在抛光斑内的序号，除去序号非数部分
                                WClnSptRmvNN=WClnSptRmv(IdxClnEvaIn);%去除函数向量化，除去序号非数部分
                                jTrj=Ser2dTrj(iy,jx);%序号
                                AConvEva(SerClnEvaInNN,jTrj)=WClnSptRmvNN;%卷积矩阵
                            end
                        end
                    end
                    WNCln0PstCal=reshape(WN2d0PstCal,NyCal*NxCal,1);%待加工误差
                    MaskClnEvaCal=reshape(MaskEvaCal,NyCal*NxCal,1);%待加工误差
                    WNCln0MaskEva=WNCln0PstCal(MaskClnEvaCal);%Mask内的待加工面形
                    DwellTimeClnMaskTrj=lsqlin(AConvEva, WNCln0MaskEva, [], [], [], [], MinDwell*ones(NNodTrj,1),...
                        MaxDwell*ones(NNodTrj,1), MinDwell*ones(NNodTrj,1));%优化算法,默认算法 interior-point
                    DwellTimeCal0=zeros(NyCal,NxCal);%驻留时间初始化
                    DwellTimeCal0(MaskTrjCal)=DwellTimeClnMaskTrj;%最小二乘法求解的驻留时间
                    DwellTimeCal0(~MaskTrjCal)=NaN;%驻留时间Trj
                    WP2dPstCal=zeros(NyCal,NxCal);%实际加工量仿真
                    for iy=NyMidCal+(-NyHalfPst:NyHalfPst)
                        for jx=NxMidCal+(-NxHalfPst:NxHalfPst)
                            if DwellTimeCal0(iy,jx)>0
                                WP2dPstCal(iy+(-NySptHalfDwn:NySptHalfUp),jx+(-NxSptHalfLft:NxSptHalfRgt))=...
                                    WP2dPstCal(iy+(-NySptHalfDwn:NySptHalfUp),jx+(-NxSptHalfLft:NxSptHalfRgt))+...
                                    DwellTimeCal0(iy,jx)*W2dSptRmv(NyMidSpt+(-NySptHalfDwn:NySptHalfUp),NxMidSpt+(-NxSptHalfLft:NxSptHalfRgt));
                            end
                        end
                    end
                    WRsd2dPstCal=W2d0PstCal-WP2dPstCal;%残差
                end
            elseif strcmp(selectedButton.Text,'均抛（等厚均抛）')==1
                WN2d0PstCal=UniformThick*ones(NyCal,NxCal);%均抛面形
                WP2dPstCal=zeros(NyCal,NxCal);%实际加工量仿真
                DwellTimeCal0=WN2d0PstCal/Impulse;%驻留时间Cal
                DwellTimeCal0(~MaskTrjCal)=NaN;%驻留时间Trj
                for iy=NyMidCal+(-NyHalfPst:NyHalfPst)
                    for jx=NxMidCal+(-NxHalfPst:NxHalfPst)
                        if DwellTimeCal0(iy,jx)>0
                            WP2dPstCal(iy+(-NySptHalfDwn:NySptHalfUp),jx+(-NxSptHalfLft:NxSptHalfRgt))=...
                                WP2dPstCal(iy+(-NySptHalfDwn:NySptHalfUp),jx+(-NxSptHalfLft:NxSptHalfRgt))+...
                                DwellTimeCal0(iy,jx)*W2dSptRmv(NyMidSpt+(-NySptHalfDwn:NySptHalfUp),NxMidSpt+(-NxSptHalfLft:NxSptHalfRgt));
                        end
                    end
                end
                WRsd2dPstCal=W2d0PstCal-WP2dPstCal;%残差
            end
            %--------------------------------------------------------------
            if exist('DwellTimeCal0','var')==1
                if strcmp(app.ApertureTabGroup.SelectedTab.Title,'圆口径')
                    TrimTrj=app.TrimTrjEditField.Value;%裁边
                    D_inner=app.InnerDiamEditField.Value;%内径
                    Dtrj_inner=max([D_inner+2*TrimTrj,0]);%无需驻留的区域
                    DwellTimeCal0(x2dCal.^2+y2dCal.^2<(Dtrj_inner/2)^2)=MinDwell;%内孔区域驻留时间最小化
                end
                app.DwellTimeCal0G=DwellTimeCal0;%存储驻留时间
                app.DwellTimeCalG=DwellTimeCal0;%存储驻留时间
                app.x2dCalG=x2dCal;%驻留时间坐标
                app.y2dCalG=y2dCal; %驻留时间坐标
                W2d0SufCal=nan(NyCal,NxCal);%零件表面区初始面型初始化
                WEvaCal=nan(NyCal,NxCal);%评价区初始面型初始化
                WRsd2dSufCal=nan(NyCal,NxCal);%零件表面区残差初始化
                WRsd2dEvaCal=nan(NyCal,NxCal);%评价区残差初始化
                W2d0SufCal(MaskSufCal)=W2d0PstCal(MaskSufCal);%零件表面区初始面型
                W2d0SufCal=W2d0SufCal-min(min(W2d0SufCal));%零件表面区面形
                WEvaCal(MaskEvaCal)=W2d0PstCal(MaskEvaCal);%评价区初始面型
                WEvaCal=WEvaCal-min(min(WEvaCal));%评价区面形
                WRsd2dSufCal(MaskSufCal)=WRsd2dPstCal(MaskSufCal);%零件表面区残差
                WRsd2dSufCal=WRsd2dSufCal-min(min(WRsd2dSufCal));%零件表面区残差
                WRsd2dEvaCal(MaskEvaCal)=WRsd2dPstCal(MaskEvaCal);%评价区残差
                WRsd2dEvaCal=WRsd2dEvaCal-min(min(WRsd2dEvaCal));%评价区残差
                WP2dPstCal(~MaskSufCal)=NaN;%实际加工量仿真，表面外非数
                %--------------------------------------------------------------
                DwellTimePerAreaCal=DwellTimeCal0/DXCal/DYCal;%单位面积驻留时间
                if isequal(size(x2dCal),size(DwellTimePerAreaCal)) && isequal(size(y2dCal),size(DwellTimePerAreaCal))
                    surf(app.DwellTimeUIAxes,x2dCal,y2dCal,DwellTimePerAreaCal);%画图
                end
                shading(app.DwellTimeUIAxes,'interp');
                axis(app.DwellTimeUIAxes,'equal');
                colorbar(app.DwellTimeUIAxes);
                view(app.DwellTimeUIAxes,[0 0 1]);
                if isequal(size(x2dCal),size(WP2dPstCal)) && isequal(size(y2dCal),size(WP2dPstCal))
                    surf(app.RemoveUIAxes,x2dCal,y2dCal,WP2dPstCal);%画图
                end
                shading(app.RemoveUIAxes,'interp');
                axis(app.RemoveUIAxes,'equal');
                colorbar(app.RemoveUIAxes);
                view(app.RemoveUIAxes,[0 0 1]);
                if isequal(size(x2dCal),size(WEvaCal)) && isequal(size(y2dCal),size(WEvaCal))
                    surf(app.SErrEvaBeforeUIAxes,x2dCal,y2dCal,WEvaCal);%画图
                end
                shading(app.SErrEvaBeforeUIAxes,'interp');
                axis(app.SErrEvaBeforeUIAxes,'equal');
                colorbar(app.SErrEvaBeforeUIAxes);
                view(app.SErrEvaBeforeUIAxes,[0 0 1]);
                if isequal(size(x2dCal),size(WRsd2dEvaCal)) && isequal(size(y2dCal),size(WRsd2dEvaCal))
                    surf(app.SErrEvaAfterUIAxes,x2dCal,y2dCal,WRsd2dEvaCal);%画图
                end
                shading(app.SErrEvaAfterUIAxes,'interp');
                axis(app.SErrEvaAfterUIAxes,'equal');
                colorbar(app.SErrEvaAfterUIAxes);
                view(app.SErrEvaAfterUIAxes,[0 0 1]);
                %--------------------------------------------------------------
                WCln0EvaCal=WEvaCal(:);%求解RMS
                WCln0EvaCal=WCln0EvaCal(~isnan(WCln0EvaCal));%求解RMS
                WCln0EvaCal=WCln0EvaCal-mean(WCln0EvaCal);%求解RMS
                WRsdClnEvaCal=WRsd2dEvaCal(:);%求解RMS
                WRsdClnEvaCal=WRsdClnEvaCal(~isnan(WRsdClnEvaCal));%求解RMS
                WRsdClnEvaCal=WRsdClnEvaCal-mean(WRsdClnEvaCal);%求解RMS
                app.RMSBeforeEvaEditField.Value=num2str(rms(WCln0EvaCal));%评价区加工前的RMS
                app.RMSAfterEvaEditField.Value=num2str(rms(WRsdClnEvaCal));%评价区加工后的RMS
                app.SolveLamp.Color=[0.00,1.00,0.00];%显示灯
            end
        end

        % Button pushed function: DwellTimeModifyButton
        function DwellTimeModifyButtonPushed(app, event)
            WidthEdge=app.WidthModifyEditField.Value;%要修正驻留时间的边缘区宽度
            HTimeEdge0=app.DwelltimeChangeEditField.Value;%边缘调整量（单位面积）
            DwellTimeCal0=app.DwellTimeCal0G;%驻留时间
            [NyCal,NxCal]=size(DwellTimeCal0);%节点数量
            NxHalfTim=(NxCal-1)/2;%半长点数
            NyHalfTim=(NyCal-1)/2;%半宽点数
            DXCal=app.DXCalEditField.Value;%面形数据间距与设置间距一致
            DYCal=app.DYCalEditField.Value;%面形数据间距与设置间距一致
            x1d=DXCal*(-NxHalfTim:1:NxHalfTim);%坐标
            y1d=DYCal*(-NyHalfTim:1:NyHalfTim);%坐标
            [x2dCal,y2dCal]=meshgrid(x1d,y1d);%坐标
            HTimeEdge=HTimeEdge0*DXCal*DYCal;%边缘调整量
            MaxDwell0=app.MaxDwellTimeEditField.Value;%驻留时间不能小于Min
            MinDwell0=app.MinDwellTimeEditField.Value;%驻留时间不能大于Max
            MaxDwell=MaxDwell0;%
            MinDwell=MinDwell0;%
            if WidthEdge~=0 && HTimeEdge~=0
                if strcmp(app.ApertureTabGroup.SelectedTab.Title,'圆口径')
                    DiamSuf=app.DiamEditField.Value;%工件边缘直径
                    Dstc2Edge=1-(DiamSuf/2-sqrt(x2dCal.^2+y2dCal.^2))/WidthEdge;%距边缘的相对距离
                    Dstc2Edge(Dstc2Edge<0)=0;%负数为0
                    Dstc2Edge(Dstc2Edge>1)=1;%超出1为1
                    D_inner=app.InnerDiamEditField.Value;%内径
                    Dstc1Edge=1-(sqrt(x2dCal.^2+y2dCal.^2)-D_inner/2)/WidthEdge;%距边缘的相对距离
                    Dstc1Edge(Dstc1Edge<0)=0;%负数为0
                    Dstc1Edge(Dstc1Edge>1)=1;%超出1为1
                    DstcEdge=1-(Dstc1Edge+Dstc2Edge);%修饰函数0-1
                    DDwellTimeCal=-HTimeEdge*DstcEdge+HTimeEdge;%驻留时间修饰量
                    DwellTimeCal=DwellTimeCal0+DDwellTimeCal;%修饰后的驻留时间
                    DwellTimeCal(DwellTimeCal<MinDwell)=MinDwell;%驻留时间不能小于Min
                    DwellTimeCal(DwellTimeCal>MaxDwell)=MaxDwell;%驻留时间不能大于Max
                elseif strcmp(app.ApertureTabGroup.SelectedTab.Title,'矩形')
                    LxSuf=app.LXEditField.Value;%Lx
                    LySuf=app.LYEditField.Value;%Ly
                    Rcor=app.RCornerEditField.Value;%圆角
                    LxEdg=LxSuf-2*WidthEdge;%LxTrj
                    LyEdg=LySuf-2*WidthEdge;%LyTrj
                    RcorEdg=max([0,Rcor-WidthEdge]);%轨迹区圆角
                    %--------------------------------------------------------------------------
                    index1=false(NyCal,NxCal);%mask标识
                    index1(abs(x2dCal)>(LxSuf/2-Rcor))=true;%位于角部
                    index2=false(NyCal,NxCal);%mask标识
                    index2(abs(y2dCal)>(LySuf/2-Rcor))=true;%位于角部
                    index3=false(NyCal,NxCal);%mask标识
                    index3((abs(x2dCal)-(LxSuf/2-Rcor)).^2+(abs(y2dCal)-(LySuf/2-Rcor)).^2>Rcor^2)=true;%位于角部且大于导圆半径
                    index4=false(NyCal,NxCal);%mask标识
                    index4(abs(x2dCal)>(LxSuf/2))=true;%位于角外部
                    index5=false(NyCal,NxCal);%mask标识
                    index5(abs(y2dCal)>(LySuf/2))=true;%位于角外部
                    MaskSufCal=~(index1&index2&index3|index4|index5);%外部区域取反为内部区域
                    %--------------------------------------------------------------------------
                    index1=false(NyCal,NxCal);%mask标识
                    index1(abs(x2dCal)>(LxEdg/2-RcorEdg))=true;%位于角部
                    index2=false(NyCal,NxCal);%mask标识
                    index2(abs(y2dCal)>(LyEdg/2-RcorEdg))=true;%位于角部
                    index3=false(NyCal,NxCal);%mask标识
                    index3((abs(x2dCal)-(LxEdg/2-RcorEdg)).^2+(abs(y2dCal)-(LyEdg/2-RcorEdg)).^2>RcorEdg^2)=true;%位于角部且大于导圆半径
                    index4=false(NyCal,NxCal);%mask标识
                    index4(abs(x2dCal)>(LxEdg/2))=true;%位于角外部
                    index5=false(NyCal,NxCal);%mask标识
                    index5(abs(y2dCal)>(LyEdg/2))=true;%位于角外部
                    MaskEdgCal=~(index1&index2&index3|index4|index5);%外部区域取反为内部区域
                    %--------------------------------------------------------------------------
                    DDwellTimeCal=nan(NyCal,NxCal);%驻留时间修饰量
                    DDwellTimeCal(MaskEdgCal)=0;%中心区不修饰
                    DDwellTimeCal(:,1)=HTimeEdge;%边缘区最高
                    DDwellTimeCal(:,end)=HTimeEdge;%边缘区最高
                    DDwellTimeCal(1,:)=HTimeEdge;%边缘区最高
                    DDwellTimeCal(end,:)=HTimeEdge;%边缘区最高
                    DDwellTimeCal(~MaskSufCal)=HTimeEdge;%边缘区最高
                    DDwellTimeCalCln=reshape(DDwellTimeCal,NyCal*NxCal,1);%向量化后剔除非数
                    x2dCalCln=reshape(x2dCal,NyCal*NxCal,1);%向量化后剔除非数
                    y2dCalCln=reshape(y2dCal,NyCal*NxCal,1);%向量化后剔除非数
                    DDwellTimeCalClnNN=DDwellTimeCalCln(~isnan(DDwellTimeCalCln));%向量化后剔除非数
                    x2dCalClnNN=x2dCalCln(~isnan(DDwellTimeCalCln));%向量化后剔除非数
                    y2dCalClnNN=y2dCalCln(~isnan(DDwellTimeCalCln));%向量化后剔除非数
                    DDwellTimeCal=griddata(x2dCalClnNN,y2dCalClnNN,DDwellTimeCalClnNN,x2dCal,y2dCal);%驻留时间修饰量
                    DwellTimeCal=DwellTimeCal0+DDwellTimeCal;%修饰后的驻留时间
                    DwellTimeCal(DwellTimeCal<MinDwell)=MinDwell;%驻留时间不能小于Min
                    DwellTimeCal(DwellTimeCal>MaxDwell)=MaxDwell;%驻留时间不能大于Max
                elseif strcmp(app.ApertureTabGroup.SelectedTab.Title,'正多边形')
                    DiamSuf=app.DiamPolyEditField.Value;%直径
                    DiamEdg=DiamSuf-WidthEdge;%边界直径
                    MaskSufCal=true(NyCal,NxCal);%mask
                    MaskEdgCal=true(NyCal,NxCal);%mask
                    N_sides=app.EdgeNumEditField.Value;%顶点数
                    N_vertx=N_sides;%顶点数
                    tht0=(2*pi)/360*app.FstVerAngNumEditField.Value;%第一角的角位
                    tht_vertx=linspace(tht0,2*pi+tht0,N_vertx+1);%顶点角度
                    x_vertexSuf=(DiamSuf/2)*cos(tht_vertx);%顶点坐标
                    y_vertexSuf=(DiamSuf/2)*sin(tht_vertx);%顶点坐标
                    x_vertexEdg=(DiamEdg/2)*cos(tht_vertx);%顶点坐标
                    y_vertexEdg=(DiamEdg/2)*sin(tht_vertx);%顶点坐标
                    for jx=1:NxCal
                        for iy=1:NyCal
                            xp=x2dCal(iy,jx);
                            yp=y2dCal(iy,jx);
                            for i_sides=1:N_sides
                                n1=[x_vertexSuf(i_sides)-xp;y_vertexSuf(i_sides)-yp;0];%点到边起点矢量
                                n2=[x_vertexSuf(i_sides+1)-x_vertexSuf(i_sides);y_vertexSuf(i_sides+1)-y_vertexSuf(i_sides);0];%边矢量
                                n3=cross(n1,n2);%叉乘矢量
                                if n3(3)<0
                                    MaskSufCal(iy,jx)=false;%叉乘矢量为负，则幅值false，则出边界，index为true是边界内，index为false是边界外
                                    break;
                                end
                            end
                            for i_sides=1:N_sides
                                n1=[x_vertexEdg(i_sides)-xp;y_vertexEdg(i_sides)-yp;0];%点到边起点矢量
                                n2=[x_vertexEdg(i_sides+1)-x_vertexEdg(i_sides);y_vertexEdg(i_sides+1)-y_vertexEdg(i_sides);0];%边矢量
                                n3=cross(n1,n2);%叉乘矢量
                                if n3(3)<0
                                    MaskEdgCal(iy,jx)=false;%叉乘矢量为负，则幅值false，则出边界，index为true是边界内，index为false是边界外
                                    break;
                                end
                            end
                        end
                    end
                    DDwellTimeCal=nan(NyCal,NxCal);%驻留时间修饰量
                    DDwellTimeCal(MaskEdgCal)=0;%驻留时间修饰量
                    DDwellTimeCal(~MaskSufCal)=HTimeEdge;%驻留时间修饰量
                    DDwellTimeCalCln=reshape(DDwellTimeCal,NyCal*NxCal,1);%向量化后剔除非数
                    x2dCalCln=reshape(x2dCal,NyCal*NxCal,1);%向量化后剔除非数
                    y2dCalCln=reshape(y2dCal,NyCal*NxCal,1);%向量化后剔除非数
                    DDwellTimeCalClnNN=DDwellTimeCalCln(~isnan(DDwellTimeCalCln));%向量化后剔除非数
                    x2dCalClnNN=x2dCalCln(~isnan(DDwellTimeCalCln));%向量化后剔除非数
                    y2dCalClnNN=y2dCalCln(~isnan(DDwellTimeCalCln));%向量化后剔除非数
                    DDwellTimeCal=griddata(x2dCalClnNN,y2dCalClnNN,DDwellTimeCalClnNN,x2dCal,y2dCal);%驻留时间修饰量
                    DwellTimeCal=DwellTimeCal0+DDwellTimeCal;%修饰后的驻留时间
                    DwellTimeCal(DwellTimeCal<MinDwell)=MinDwell;%驻留时间不能小于Min
                    DwellTimeCal(DwellTimeCal>MaxDwell)=MaxDwell;%驻留时间不能大于Max
                end
            else
                DwellTimeCal=app.DwellTimeCal0G;%驻留时间
            end
            app.DwellTimeCalG=DwellTimeCal;%存储驻留时间
            app.x2dCalG=x2dCal;%驻留时间坐标
            app.y2dCalG=y2dCal; %驻留时间坐标
            DwellTimePerAreaCal=DwellTimeCal/DXCal/DYCal;%单位面积驻留时间
            if isequal(size(x2dCal),size(DwellTimePerAreaCal)) && isequal(size(y2dCal),size(DwellTimePerAreaCal)) && ~all(all(isnan(DwellTimePerAreaCal)))
                pcolor(app.DwellTimeUIAxes,x2dCal,y2dCal,DwellTimePerAreaCal);%画图
                shading(app.DwellTimeUIAxes,'interp');
                axis(app.DwellTimeUIAxes,'equal');
                colorbar(app.DwellTimeUIAxes);
                app.SolveLamp.Color=[0.00,1.00,0.00];%显示灯
            end
        end

        % Button pushed function: TrajButton
        function TrajButtonPushed(app, event)
            xcln0=app.xmld0G;%模型坐标
            ycln0=app.ymld0G;%模型坐标
            zcln0=app.zmld0G;%模型坐标
            nxcln0=app.nxmld0G;%模型坐标
            nycln0=app.nymld0G;%模型坐标
            nzcln0=app.nzmld0G;%模型坐标
            if strcmp(app.TrajTabGroup.SelectedTab.Title,'栅线')
                DXTrj=app.DXTrajParallelEditField.Value;%X间距
                DYTrj=app.DYTrajParallelEditField.Value;%Y间距
                LineDirect = app.LineDirectParallelDropDown.Value;%栅线方向
                StartPoint = app.StartPointParallelDropDown.Value;%起点位置
                if strcmp(app.ApertureTabGroup.SelectedTab.Title,'圆口径')
                    DSuf=app.DiamEditField.Value;%零件表面直径
                    TrimTrj=app.TrimTrjEditField.Value;%轨迹区裁边
                    DTrj=DSuf-2*TrimTrj;%轨迹区直径
                    Nhalfx=round(DTrj/2/DXTrj);%一半节点数
                    Nhalfy=round(DTrj/2/DYTrj);%一半节点数
                elseif strcmp(app.ApertureTabGroup.SelectedTab.Title,'矩形')
                    LxSuf=app.LXEditField.Value;%Lx
                    LySuf=app.LYEditField.Value;%Ly
                    TrimTrj=app.TrimTrjEditField.Value;%轨迹区裁边
                    LxTrj=LxSuf-2*TrimTrj;%轨迹区长度
                    LyTrj=LySuf-2*TrimTrj;%轨迹区宽度
                    Nhalfx=round(LxTrj/2/DXTrj);%一半节点数
                    Nhalfy=round(LyTrj/2/DYTrj);%一半节点数
                elseif strcmp(app.ApertureTabGroup.SelectedTab.Title,'正多边形')
                    DiamSuf=app.DiamPolyEditField.Value;%直径
                    TrimTrj=app.TrimTrjEditField.Value;%轨迹区裁边
                    DTrj=DiamSuf-2*TrimTrj;%轨迹区直径
                    Nhalfx=round(DTrj/2/DXTrj);%一半节点数
                    Nhalfy=round(DTrj/2/DYTrj);%一半节点数
                end
                x1d=(-Nhalfx:1:Nhalfx)*DXTrj;%坐标
                y1d=(-Nhalfy:1:Nhalfy)*DYTrj;%坐标
                Nx=length(x1d);%节点数量
                Ny=length(y1d);%节点数量
                [x2d,y2d]=meshgrid(x1d,y1d);%二维坐标
                if strcmp(StartPoint,'左下')
                    x2dtemp1=x2d;%暂存
                    y2dtemp1=y2d;%暂存
                elseif strcmp(StartPoint,'右下')
                    x2dtemp1=fliplr(x2d);%暂存
                    y2dtemp1=fliplr(y2d);%暂存
                elseif strcmp(StartPoint,'右上')
                    x2dtemp1=rot90(x2d,2);%暂存
                    y2dtemp1=rot90(y2d,2);%暂存
                elseif strcmp(StartPoint,'左上')
                    x2dtemp1=flipud(x2d);%暂存
                    y2dtemp1=flipud(y2d);%暂存
                end
                x2dtemp2=x2dtemp1;%暂存
                y2dtemp2=y2dtemp1;%暂存
                if strcmp(LineDirect,'X')
                    x2dtemp2(2:2:end,:)=fliplr(x2dtemp1(2:2:end,:));%返程反向
                    y2dtemp2(2:2:end,:)=fliplr(y2dtemp1(2:2:end,:));%返程反向
                    x_trj0=reshape(x2dtemp2',Nx*Ny,1);%2维坐标列向量化
                    y_trj0=reshape(y2dtemp2',Nx*Ny,1);%2维坐标列向量化
                elseif strcmp(LineDirect,'Y')
                    x2dtemp2(:,2:2:end)=flipud(x2dtemp1(:,2:2:end));%返程反向
                    y2dtemp2(:,2:2:end)=flipud(y2dtemp1(:,2:2:end));%返程反向
                    x_trj0=reshape(x2dtemp2,Nx*Ny,1);%2维坐标列向量化
                    y_trj0=reshape(y2dtemp2,Nx*Ny,1);%2维坐标列向量化
                end
            elseif strcmp(app.TrajTabGroup.SelectedTab.Title,'螺旋线')&&strcmp(app.ApertureTabGroup.SelectedTab.Title,'圆口径')
                DSuf=app.DiamEditField.Value;%零件表面直径
                dAzi=app.dAzimuthEditField_2.Value;%间距
                dR=app.dREditField_2.Value;%间距
                StartPoint0 = app.StartPointAngleEditField_2.Value;%起点位置
                StartPoint=StartPoint0/360*2*pi;%初始角度
                Ncircle=floor(DSuf/2/dR);%圈数
                Ncnod=100;%每圈点数
                if strcmp(app.RotDirectDropDown_2.Value,'顺时针')
                    tht0=linspace(2*pi*Ncircle,0,Ncircle*Ncnod+1);%节点数量
                else
                    tht0=linspace(0,2*pi*Ncircle,Ncircle*Ncnod+1);%节点数量
                end
                if strcmp(app.RadDirectDropDown_2.Value,'内向外')
                    rp0=linspace(0,DSuf/2,Ncircle*Ncnod+1);%节点数量
                else
                    rp0=linspace(DSuf/2,0,Ncircle*Ncnod+1);%节点数量
                end
                x0=rp0.*cos(tht0+StartPoint);%螺旋线
                y0=rp0.*sin(tht0+StartPoint);%螺旋线
                s0=zeros(size(x0));%弧长坐标初始化
                for i0=2:length(x0)
                    s0(i0)=s0(i0-1)+sqrt((x0(i0)-x0(i0-1))^2+(y0(i0)-y0(i0-1))^2);%弧长坐标
                end
                Ls=s0(end);%总弧长
                n_trj0=round(Ls/dAzi)+1;%插值后弧长坐标点数
                s_trj0=linspace(0,Ls,n_trj0)';%弧长坐标
                x_trj0=interp1(s0,x0,s_trj0);%螺旋线轨迹坐标
                y_trj0=interp1(s0,y0,s_trj0);%螺旋线轨迹坐标
            elseif strcmp(app.TrajTabGroup.SelectedTab.Title,'同心圆')&&strcmp(app.ApertureTabGroup.SelectedTab.Title,'圆口径')
                DSuf=app.DiamEditField.Value;%零件表面直径
                dAzi=app.dAzimuthEditField_3.Value;%间距
                dR=app.dREditField_3.Value;%间距
                StartPoint0 = app.StartPointAngleEditField_3.Value;%起点位置
                StartPoint=StartPoint0/360*2*pi;%初始角度
                Ncircle=ceil(DSuf/2/dR);%圈数
                if strcmp(app.RadDirectDropDown_3.Value,'内向外')
                    rp0=linspace(0,DSuf/2,Ncircle+1);%节点数量
                else
                    rp0=linspace(DSuf/2,0,Ncircle+1);%节点数量
                end
                x_trj0=[];%初始化
                y_trj0=[];%初始化
                for iR=1:length(rp0)
                    Ncnod=round(2*pi*rp0(iR)/dAzi);%每圈点数
                    if strcmp(app.RotDirectDropDown_3.Value,'顺时针')
                        tht0=linspace(2*pi,0,Ncnod+1);%节点数量
                    else
                        tht0=linspace(0,2*pi,Ncnod+1);%节点数量
                    end
                    x_trjt=rp0(iR).*cos(tht0+StartPoint);%一圈数数据
                    y_trjt=rp0(iR).*sin(tht0+StartPoint);%一圈数数据
                    if Ncnod<1 %一圈只有一个点时
                        x_trj0=[x_trj0,x_trjt];
                        y_trj0=[y_trj0,y_trjt];
                    else
                        x_trj0=[x_trj0,x_trjt(2:end)];
                        y_trj0=[y_trj0,y_trjt(2:end)];
                    end
                end
            elseif strcmp(app.TrajTabGroup.SelectedTab.Title,'矩形螺旋线')&&strcmp(app.ApertureTabGroup.SelectedTab.Title,'矩形')
                DXTrj=app.DXTrajEditField_4.Value;%X间距
                DYTrj=app.DYTrajEditField_4.Value;%Y间距
                StartPoint = app.StartPointDropDown_4.Value;%起点位置
                LxSuf=app.LXEditField.Value;%Lx
                LySuf=app.LYEditField.Value;%Ly
                Nhalfx=floor((LxSuf-0.01)/2/DXTrj);%一半节点数
                Nhalfy=floor((LySuf-0.01)/2/DYTrj);%一半节点数
                x1d=(-Nhalfx:1:Nhalfx)*DXTrj;%坐标
                y1d=(-Nhalfy:1:Nhalfy)*DYTrj;%坐标
                Nx=length(x1d);%节点数量
                Ny=length(y1d);%节点数量
                [x2d,y2d]=meshgrid(x1d,y1d);%二维坐标
                if strcmp(StartPoint,'左下')
                    x2dtemp1=x2d;%暂存
                    y2dtemp1=y2d;%暂存
                elseif strcmp(StartPoint,'右下')
                    x2dtemp1=fliplr(x2d);%暂存
                    y2dtemp1=fliplr(y2d);%暂存
                elseif strcmp(StartPoint,'右上')
                    x2dtemp1=rot90(x2d,2);%暂存
                    y2dtemp1=rot90(y2d,2);%暂存
                elseif strcmp(StartPoint,'左上')
                    x2dtemp1=flipud(x2d);%暂存
                    y2dtemp1=flipud(y2d);%暂存
                end
                x2d=x2dtemp1;%暂存
                y2d=y2dtemp1;%暂存
                x_trj0=[];%轨迹初始化
                y_trj0=[];%轨迹初始化
                for icyc=1:max([Nx,Ny])
                    x_trj0=[x_trj0;x2d(icyc,icyc:Nx-(icyc-1))'];%四边形轨迹
                    y_trj0=[y_trj0;y2d(icyc,icyc:Nx-(icyc-1))'];%四边形轨迹
                    if length(x_trj0)>=Ny*Nx %判断是否完成所有点
                        break;
                    end
                    x_trj0=[x_trj0;x2d(icyc+1:Ny-(icyc-1),Nx-(icyc-1))];%四边形轨迹
                    y_trj0=[y_trj0;y2d(icyc+1:Ny-(icyc-1),Nx-(icyc-1))];%四边形轨迹
                    if length(x_trj0)>=Ny*Nx %判断是否完成所有点
                        break;
                    end
                    x_trj0=[x_trj0;fliplr(x2d(Ny-(icyc-1),icyc:Nx-icyc))'];%四边形轨迹
                    y_trj0=[y_trj0;fliplr(y2d(Ny-(icyc-1),icyc:Nx-icyc))'];%四边形轨迹
                    if length(x_trj0)>=Ny*Nx %判断是否完成所有点
                        break;
                    end
                    x_trj0=[x_trj0;flipud(x2d(icyc+1:Ny-icyc,icyc))];%四边形轨迹
                    y_trj0=[y_trj0;flipud(y2d(icyc+1:Ny-icyc,icyc))];%四边形轨迹
                    if length(x_trj0)>=Ny*Nx %判断是否完成所有点
                        break;
                    end
                end
                if strcmp(app.RadDirectDropDown_4.Value,'内向外')
                    x_trj0=flipud(x_trj0);%反向，由内向外
                    y_trj0=flipud(y_trj0);%反向，由内向外
                end
            else
                x_trj0=[];
                y_trj0=[];
            end
            %生成面形轨迹点-------------------------------------------------------------
            DwellTimeCal=app.DwellTimeCalG;%存储驻留时间
            x2dCal=app.x2dCalG;%驻留时间坐标
            y2dCal=app.y2dCalG; %驻留时间坐标
            DwellTime_trj0=griddata(x2dCal,y2dCal,DwellTimeCal,x_trj0,y_trj0);%驻留时间

                    (strcmp(app.ApertureTabGroup.SelectedTab.Title,'圆口径')&&strcmp(app.TrajTabGroup.SelectedTab.Title,'同心圆'))

                DwellTime_trj0(x_trj0.^2+y_trj0.^2<(Dtrj_inner/2)^2)=NaN;%非数
            end
            x_trj=x_trj0(~isnan(DwellTime_trj0));%剔除非数
            y_trj=y_trj0(~isnan(DwellTime_trj0));%剔除非数
            z_trj=griddata(xcln0,ycln0,zcln0,x_trj,y_trj);%去除边界外的坐标
            nxclnt=griddata(xcln0,ycln0,nxcln0,x_trj,y_trj);%去除边界外的坐标
            nyclnt=griddata(xcln0,ycln0,nycln0,x_trj,y_trj);%去除边界外的坐标
            nzclnt=griddata(xcln0,ycln0,nzcln0,x_trj,y_trj);%去除边界外的坐标
            nx_trj=nxclnt./sqrt(nxclnt.^2+nyclnt.^2+nzclnt.^2);%单位化
            ny_trj=nyclnt./sqrt(nxclnt.^2+nyclnt.^2+nzclnt.^2);%单位化
            nz_trj=nzclnt./sqrt(nxclnt.^2+nyclnt.^2+nzclnt.^2);%单位化
            x_trj=reshape(x_trj,numel(x_trj),1);%列向量化，方便存储
            y_trj=reshape(y_trj,numel(y_trj),1);%列向量化，方便存储
            z_trj=reshape(z_trj,numel(z_trj),1);%列向量化，方便存储
            nx_trj=reshape(nx_trj,numel(nx_trj),1);%列向量化，方便存储
            ny_trj=reshape(ny_trj,numel(ny_trj),1);%列向量化，方便存储
            nz_trj=reshape(nz_trj,numel(nz_trj),1);%列向量化，方便存储
            if isequal(size(x_trj),size(z_trj)) && isequal(size(y_trj),size(z_trj)) && ~isempty(x_trj)
                plot(app.Traj2DUIAxes,x_trj,y_trj,x_trj(1),y_trj(1),'s');
                axis(app.Traj2DUIAxes,'equal');%画图
                plot3(app.Traj3DUIAxes,x_trj,y_trj,z_trj,x_trj(1),y_trj(1),z_trj(1),'s');
            else
                cla(app.Traj2DUIAxes);
                cla(app.Traj3DUIAxes);
            end
            app.xTrjG=x_trj;%保存轨迹
            app.yTrjG=y_trj;%保存轨迹
            app.zTrjG=z_trj;%保存轨迹
            app.nxTrjG=nx_trj;%保存轨迹
            app.nyTrjG=ny_trj;%保存轨迹
            app.nzTrjG=nz_trj;%保存轨迹
            app.CNCLamp.Color=[1.00,0.00,0.00];%显示灯
        end

        % Button pushed function: CNCButton
        function CNCButtonPushed(app, event)
            if strcmp(app.ToolDropDown.Value,'小工具')
                app.RToolEditField.Value=0;%球头半径按0处理
                app.XAngleEditField.Value=0;%工具转角为0
                app.YAngleEditField.Value=0;%工具转角为0
            end
            DwellTimeCal=app.DwellTimeCalG;%存储驻留时间
            DXCal=app.DXCalG;%计算用的节点间距
            DYCal=app.DYCalG;%计算用的节点间距
            dx=app.DXTrajParallelEditField.Value;%X间距
            dy=app.DYTrajParallelEditField.Value;%Y间距
            DwellTimePerAreaCal=DwellTimeCal/DXCal/DYCal;%驻留时间分布
            DwellTimeEachPointTrj=DwellTimePerAreaCal*dx*dy;%轨迹点驻留时间
            x2dCal=app.x2dCalG;%原始数据坐标
            y2dCal=app.y2dCalG;%原始数据坐标
            x_trj0=app.xTrjG;%保存轨迹
            y_trj0=app.yTrjG;%保存轨迹
            z_trj0=app.zTrjG;%保存轨迹
            nx_trj0=app.nxTrjG;%保存轨迹
            ny_trj0=app.nyTrjG;%保存轨迹
            nz_trj0=app.nzTrjG;%保存轨迹
            DwellTimeTrj0=griddata(x2dCal,y2dCal,DwellTimeEachPointTrj,x_trj0,y_trj0,'linear');%插值获取轨迹点驻留时间
            DwellTimeTrj0=DwellTimeTrj0(~isnan(DwellTimeTrj0));%保存轨迹
            x_trj=x_trj0(~isnan(DwellTimeTrj0));%保存轨迹
            y_trj=y_trj0(~isnan(DwellTimeTrj0));%保存轨迹
            z_trj=z_trj0(~isnan(DwellTimeTrj0));%保存轨迹
            nx_trj=nx_trj0(~isnan(DwellTimeTrj0));%保存轨迹
            ny_trj=ny_trj0(~isnan(DwellTimeTrj0));%保存轨迹
            nz_trj=nz_trj0(~isnan(DwellTimeTrj0));%保存轨迹
            DwellTimeTrj=DwellTimeTrj0./nz_trj;%考虑曲面面积变化的单点驻留时间
            alpx_tool_rad0=app.XAngleEditField.Value/360*2*pi;%角度值转弧度值
            bety_tool_rad0=app.YAngleEditField.Value/360*2*pi;%角度值转弧度值
            Rtool=app.RToolEditField.Value;%球头半径
            depth_cut=app.CylinderCompEditField.Value;%气缸压入量
            offset_x=app.XoffsetEditField.Value;%x偏置
            offset_y=app.YoffsetEditField.Value;%y偏置
            offset_z=app.ZoffsetEditField.Value;%z偏置
            Talpx=[1,0,0;0,cos(alpx_tool_rad0),-sin(alpx_tool_rad0);0,sin(alpx_tool_rad0),cos(alpx_tool_rad0)];%绕X旋转矩阵
            Tbety=[cos(bety_tool_rad0),0,sin(bety_tool_rad0);0,1,0;-sin(bety_tool_rad0),0,cos(bety_tool_rad0)];%绕Y旋转矩阵
            ntool0=[0;0;1];%工具竖直的姿态
            nsurf0=[0;0;1];%表面向上的垂直法向量
            ntoolsuf0=Tbety*Talpx*ntool0;%工具在水平面抛光的姿态（两个切向均水平）
            for i_nod=1:length(x_trj)
                p1=nsurf0;%表面向上的垂直法向量
                p2=[nx_trj(i_nod);ny_trj(i_nod);nz_trj(i_nod)];%表面法向量
                n0=cross(p1,p2);%旋转轴向量
                if norm(n0)==0
                    n=n0;
                else 
                    n=n0/norm(n0);%单位化
                end
                tht=acos(sum(p1.*p2)/(norm(p1)*norm(p2)));%竖直姿态到表面法向量的旋转角度
                Rot=[cos(tht)+n(1)^2*(1-cos(tht)),...
                    n(1)*n(2)*(1-cos(tht))-n(3)*sin(tht),...
                    n(2)*sin(tht)+n(1)*n(3)*(1-cos(tht));
                    n(3)*sin(tht)+n(1)*n(2)*(1-cos(tht)),...
                    cos(tht)+n(2)^2*(1-cos(tht)),...
                    -n(1)*sin(tht)+n(2)*n(3)*(1-cos(tht));
                    -n(2)*sin(tht)+n(1)*n(3)*(1-cos(tht)),...
                    n(1)*sin(tht)+n(2)*n(3)*(1-cos(tht)),...
                    cos(tht)+n(3)^2*(1-cos(tht))];
                ntool_abs=Rot*ntoolsuf0;%绝对坐标系工具姿态
                nxtool_abs(i_nod,1)=ntool_abs(1);%绝对坐标系工具姿态
                nytool_abs(i_nod,1)=ntool_abs(2);%绝对坐标系工具姿态
                nztool_abs(i_nod,1)=ntool_abs(3);%绝对坐标系工具姿态
                alpx_tool_ang(i_nod,1)=asind(-nytool_abs(i_nod,1)/sqrt(nxtool_abs(i_nod,1).^2+nytool_abs(i_nod,1).^2+nztool_abs(i_nod,1).^2));%工具绕轴角
                bety_tool_ang(i_nod,1)=asind(nxtool_abs(i_nod,1)/sqrt(nxtool_abs(i_nod,1).^2+nztool_abs(i_nod,1).^2));%工具绕Y轴角
            end
            if ~isempty(x_trj) && ~isempty(y_trj) && ~isempty(z_trj) && ~isempty(nxtool_abs) && ~isempty(nytool_abs) && ~isempty(nztool_abs)
                x_path=x_trj+Rtool*nx_trj-nxtool_abs*depth_cut+offset_x;%减去吃刀量，再加上偏置量
                y_path=y_trj+Rtool*ny_trj-nytool_abs*depth_cut+offset_y;%减去吃刀量，再加上偏置量
                z_path=z_trj+Rtool*nz_trj-Rtool-nztool_abs*depth_cut+offset_z;%减去吃刀量，再加上偏置量
                lx=abs(x_path(2:end)-x_path(1:end-1));%x距离
                ly=abs(y_path(2:end)-y_path(1:end-1));%y距离
                lz=abs(z_path(2:end)-z_path(1:end-1));%z距离
                lxyz=sqrt(lx.^2+ly.^2+lz.^2);%距离
                feedspeed0=zeros(size(z_path));%进给速度
                for ip=1:length(lxyz)
                    if lx(ip,1)>(dx/dy)*ly(ip,1)
                        feedspeed0(ip+1)=2*lxyz(ip)*dx/lx(ip)/(DwellTimeTrj(ip+1)+DwellTimeTrj(ip));%进给速度
                    else
                        feedspeed0(ip+1)=2*lxyz(ip)*dy/ly(ip)/(DwellTimeTrj(ip+1)+DwellTimeTrj(ip));%进给速度
                    end
                end
                feedspeed0(1)=feedspeed0(2);%第一点ip+1，1号位根本就没算
                feedspeed=feedspeed0;%暂存
                feedspd_lwst=app.MinSpeedEditField.Value;%限速
                feedspd_fast=app.MaxSpeedEditField.Value;%限速
                feedspeed(feedspeed0<feedspd_lwst)=feedspd_lwst;%最小速度限制
                feedspeed(feedspeed0>feedspd_fast)=feedspd_fast;%最大速度限制
                total_polishtime=sum(lxyz./feedspeed(2:end));%总时间
                if isequal(size(x_path),size(z_path)) && isequal(size(y_path),size(z_path)) && ~isempty(x_path)
                    plot3(app.FeedSpeedIdealUIAxes,x_path,y_path,feedspeed0,x_path(1),y_path(1),feedspeed0(1),'s');%理论进给速度
                    plot3(app.FeedSpeedUIAxes,x_path,y_path,feedspeed,x_path(1),y_path(1),feedspeed(1),'s');%限速后进给速度
                    plot(app.Traj2DUIAxes,x_path,y_path,x_path(1),y_path(1),'s');%TCP点轨迹
                    plot3(app.Traj3DUIAxes,x_path,y_path,z_path,x_path(1),y_path(1),z_path(1),'s');%TCP点轨迹
                else
                    cla(app.FeedSpeedIdealUIAxes);
                    cla(app.FeedSpeedUIAxes);
                    cla(app.Traj2DUIAxes);
                    cla(app.Traj3DUIAxes);
                end
                %--------------------------------------------------------------------------
                data=zeros(4+length(x_trj)+1,6);%轨迹文件生成
                data(4+(1:length(x_trj)),1:6)=[x_path,y_path,z_path,alpx_tool_ang,bety_tool_ang,feedspeed];%轨迹文件
                data(4+length(x_trj)+1,1:6)=[x_path(end),y_path(end),z_path(end)+2*depth_cut+20,alpx_tool_ang(end),bety_tool_ang(end),feedspd_fast/2];%轨迹文件
                data(1,1:6)=[x_path(1),    y_path(1),  z_path(1)+100,  alpx_tool_ang(1),  bety_tool_ang(1),feedspd_fast/2];%出库开始点
                data(2,1:6)=[x_path(1),    y_path(1),      z_path(1),  alpx_tool_ang(1),  bety_tool_ang(1),feedspd_fast/2];%开始点
                data(3,1:6)=data(4+length(x_trj)+1,1:6);%结束点
                data(4,1:6)=[x_path(end),y_path(end),z_path(end)+100,alpx_tool_ang(end),bety_tool_ang(end),feedspd_fast/2];%结束泊车点
                app.PolishingTime.Value=num2str(total_polishtime);%总抛光时间
                app.dataCNCG=data;%保存
            end
        end

        % Button pushed function: ToolConButton
        function ToolConButtonPushed(app, event)
            c=-1/app.REditField.Value;%c
            K=app.KEditField.Value;
            A=[app.A4EditField.Value app.A6EditField.Value app.A8EditField.Value];
            y=0;
            for x=0:1:app.DEditField.Value
                 
                dzdx =-((2*c*x)/((- (x^2 + y^2)*(K + 1)*c^2 + 1)^(1/2) + 1) + (c^3*x*(x^2 + y^2)*(K + 1))/((1 - c^2*(x^2 + y^2)*(K + 1))^(1/2)*((1 - c^2*(x^2 + y^2)*(K + 1))^(1/2) + 1)^2));
                dzdy =-((2*c*y)/((- (x^2 + y^2)*(K + 1)*c^2 + 1)^(1/2) + 1) + (c^3*y*(x^2 + y^2)*(K + 1))/((1 - c^2*(x^2 + y^2)*(K + 1))^(1/2)*((1 - c^2*(x^2 + y^2)*(K + 1))^(1/2) + 1)^2));
                for iA=1:length(A)
	                dzdx=dzdx-2*A(iA)*(iA+1)*x*(x.^2+y.^2).^iA    
	                dzdy=dzdy-2*A(iA)*(iA+1)*y*(x.^2+y.^2).^iA
                end
            if x==0
                 r=app.REditField.Value;
            else
                k=dzdx;
                r=sqrt(x^2+y^2)/sin(atan(k));%%%点X处等效曲率半径   ( X=0,seita=0,r=R )
            end
            s=r-sqrt(r^2-(app.ToolCaliberEditField.Value/2)^2)%非球面与抛光盘最大间隙（一般须优于0.05mm）
            s=abs(round(s*10^6)/10^6);
            S(x+1)=s;%数据存储
            end
            %工具盘与工件不同位置接触间隙
            x=0:1:app.DEditField.Value;
            plot(app.ToolAdaptUIAxes,x,S);%无点绘图
            shading(app.ToolAdaptUIAxes,"interp");%插值绘图
            app.ToolAdaptUIAxes.YLim=[0,0.2];%插值绘图
            app.ToolAdaptUIAxes.XLim=[0,app.DEditField.Value/2];%插值绘图
            for i=1:1:length(S)-1
                s_speed=S(i+1)-S(i);
                s_speed=abs(round(s_speed*10^6)/10^6);
                S_speed(i)=s_speed;

            end
            %工具盘与工件不同位置接触间隙速度变化
            x=0:1:app.DEditField.Value-1;
            plot(app.ToolAdaptSpeedUIAxes,x,S_speed);%无点绘图
            shading(app.ToolAdaptSpeedUIAxes,"interp");%插值绘图
            app.ToolAdaptSpeedUIAxes.YLim=[0,5*10^-5];%插值绘图
            app.ToolAdaptSpeedUIAxes.XLim=[0,app.DEditField.Value/2];%插值绘图
        end
